class Window:
    """UTC half-open `[start, end)`."""

    start: datetime
    end: datetime

    def __post_init__(self) -> None:

        object.__setattr__(self, "end", utc(self.end))

def day_windows(start: datetime, end: datetime, bracket: timedelta = BRACKET) -> list[Window]:
    """UTC-day windows intersecting `[start - bracket, end + bracket)`."""

def _overlaps(a: tuple[datetime, datetime], b: tuple[datetime, datetime]) -> bool:
    return a[0] <= b[1] and b[0] <= a[1]


def select_cohort(readings: tuple[AcceptedReading, ...], window: Window,
                  explicit: tuple[CohortMember, ...] | None = None) -> CohortSelection:
    """`readings` cover the read range (window plus bracket). Only streams with a reading inside
    the window itself are candidates; a stream seen only in the bracket is not pulled in."""
    present = sorted({r.member for r in readings if window.contains(r.placed_at)})
    if explicit is not None:
        chosen = tuple(sorted(set(explicit)))
        unknown = [m for m in chosen if m not in present]
        reason = None if not unknown else (
            "explicit cohort names stream(s) with no accepted reading in the window: "
            + ", ".join(f"{m.source_id}@{m.sequence_epoch}" for m in unknown))
        mine = tuple(r for r in readings if r.member in set(chosen))
        others = tuple(m for m in present if m not in set(chosen))
        clash = _clashing(mine)
        if clash:
            return CohortSelection("AMBIGUOUS", (), alternatives=tuple(sorted(clash)),
                                   reason="the explicit cohort names overlapping epochs of one source")
        return CohortSelection("SELECTED_EXPLICIT", chosen, alternatives=others,
                               limits=series_limits(mine), reason=reason)
    if not present:
        return CohortSelection("EMPTY", (), reason="no accepted reading in the window")
    # Streams seen only in the bracket are not candidates and are not named: naming them would
    # change this window's basis whenever a neighbour day receives evidence.
    readings = tuple(r for r in readings if r.member in set(present))
    clashing = _clashing(readings)
    if clashing:
        names = sorted(clashing)
        return CohortSelection(
            "AMBIGUOUS", (), alternatives=tuple(names),
            reason=("a source reports the same time span under more than one sequence epoch ("
                    + ", ".join(f"{m.source_id}@{m.sequence_epoch}" for m in names)
                    + "); no coherent default cohort; select one explicitly"))
    successive = _multi_epoch(present)
    if successive:
        names = sorted(successive)
        return CohortSelection(
            "AMBIGUOUS", (), alternatives=tuple(names),
            reason=("a source reports in this window under more than one sequence epoch ("
                    + ", ".join(f"{m.source_id}@{m.sequence_epoch}" for m in names)
                    + "); central evidence does not attest that they are one physical stream, so the "
                      "default cohort does not blend them; select one explicitly"))
    return CohortSelection("SELECTED_UNIQUE", tuple(present), limits=series_limits(readings))


def _multi_epoch(members: list[CohortMember]) -> set[CohortMember]:
    """Members of a source (same gateway and source) present under two or more epochs."""
    by_source: dict[tuple, list[CohortMember]] = defaultdict(list)
    for m in members:
        by_source[(m.gateway_id, m.source_id)].append(m)
    return {m for group in by_source.values() if len(group) > 1 for m in group}


def _clashing(readings: tuple[AcceptedReading, ...]) -> set[CohortMember]:
    """Members of one source (same gateway and source) whose source-time spans overlap."""
    spans: dict[CohortMember, tuple[datetime, datetime]] = {}
    for r in readings:
        lo, hi = spans.get(r.member, (r.placed_at, r.placed_at))
        spans[r.member] = (min(lo, r.placed_at), max(hi, r.placed_at))
    by_source: dict[tuple, list[CohortMember]] = defaultdict(list)
    for m in sorted(spans):
        by_source[(m.gateway_id, m.source_id)].append(m)
    clashing: set[CohortMember] = set()
    for members in by_source.values():
        for i, a in enumerate(members):
            for b in members[i + 1:]:
                if _overlaps(spans[a], spans[b]):
                    clashing.update((a, b))
    return clashing

# Architect answers: LR-01 source-time cohort and money limit (2026-10-15)

Authority: `tasks/LR-01-late-reading-series-and-money-limit.md` Q1-Q4, `D-2026-10-14-s402c-late-sequence-and-s5-money`, `.agent/architect-S4-02c-answers.md`, and `.agent/S4-02c-review.md` F1.

## Q1 - clock proof and limit

**Accept.** Source-time ordering requires a durable, versioned, tenant/Site/source/epoch-scoped attestation from an identified authority, valid for every reading used, with a proven maximum UTC error no greater than 60 seconds. Widen each source or measurement interval by that error. A simulated attestation is explicitly identified and cannot validate a real source; `source_mode` alone is insufficient. Missing, expired, conflicting, or clock-step-affected proof leaves the series limited. Pin the exact attestation version in the new analytic basis; do not alter old runs. The 60-second bound is a ceiling, not a gateway capability claim.

## Q2 - overlap and epoch

**Accept.** Equal or overlapping uncertainty intervals in one channel remain `REPORTS_OVERLAP` and ambiguous. Gateway sequence resolves identity and duplicates, never an overlapping measurement or source-time order. Do not infer continuity across epochs from timestamps. Retain the current ambiguous default and explicit cohort-selection behavior; any automatic epoch continuity requires a separately reviewed public proof contract.

## Q3 - affected windows

**Accept with scope clarification.** On attestation creation, replacement, expiry, or withdrawal, find accepted readings whose clock basis could change, including readings covered by the old or new validity interval. Enqueue every UTC-day window whose six-hour read bracket intersects their source or measurement intervals, plus the neighboring continuity windows reached by the existing job expansion. Use the same bracket for query and requeue. If either bound changes, version and revise both together. Reassess through existing scoped jobs and append-only selections; preserve old IDs and idempotent replay.

## Q4 - Finding and eligibility

**Accept.** Keep `SUPPORTED` only for proven Candidate spans. A material series refusal that truncates the episode yields a new FindingRevision with typed `GENERATOR_SERIES_AMBIGUOUS`, refused span and evidence/assessment links, and `SERIES_LIMIT` as end reason; retain `OPEN_BOUND` only if independently true. Derive the versioned three-state `money_eligibility` from the latest selected window and latest Finding on each read. `PENDING_REASSESSMENT` wins while relevant S3/S4 work or selection staleness remains; after work settles, `WITHHELD_GENERATOR_SERIES` applies to a material unresolved refusal. `ELIGIBLE` passes this gate only: an amount still requires the exact-head `CURRENT PRICED` consequence and existing source/currentness checks. Both S4 Financials and S5 consume the same state and exact lineage. Preserve the recorded 7.5625 USD revision as history without a current-amount label.

## Packet changes for the Planner

1. Make attestation-change requeue cover both old and new validity intervals, affected accepted reading/measurement intervals, and the same six-hour query bracket; name creation, replacement, expiry, and withdrawal in phase 1 proof.
2. Specify the eligibility precedence and that `ELIGIBLE` is only the series/reassessment gate, not a substitute for `CURRENT PRICED` and the existing source/currentness checks. Assert pending, settled-withheld, and eligible cases in phase 2 proof.
3. Make explicit that an epoch may use the existing explicit-selection path, while automatic continuity across epochs is outside LR-01.

+# Architect answers: LR-01 source-time cohort and money limit (2026-10-15)
+
+Authority: `tasks/LR-01-late-reading-series-and-money-limit.md` Q1-Q4, `D-2026-10-14-s402c-late-sequence-and-s5-money`, `.agent/architect-S4-02c-answers.md`, and `.agent/S4-02c-review.md` F1.
+
+## Q1 - clock proof and limit
+
+**Accept.** Source-time ordering requires a durable, versioned, tenant/Site/source/epoch-scoped attestation from an identified authority, valid for every reading used, with a proven maximum UTC error no greater than 60 seconds. Widen each source or measurement interval by that error. A simulated attestation is explicitly identified and cannot validate a real source; `source_mode` alone is insufficient. Missing, expired, conflicting, or clock-step-affected proof leaves the series limited. Pin the exact attestation version in the new analytic basis; do not alter old runs. The 60-second bound is a ceiling, not a gateway capability claim.
+
+## Q2 - overlap and epoch
+
+**Accept.** Equal or overlapping uncertainty intervals in one channel remain `REPORTS_OVERLAP` and ambiguous. Gateway sequence resolves identity and duplicates, never an overlapping measurement or source-time order. Do not infer continuity across epochs from timestamps. Retain the current ambiguous default and explicit cohort-selection behavior; any automatic epoch continuity requires a separately reviewed public proof contract.
+
+## Q3 - affected windows
+
+**Accept with scope clarification.** On attestation creation, replacement, expiry, or withdrawal, find accepted readings whose clock basis could change, including readings covered by the old or new validity interval. Enqueue every UTC-day window whose six-hour read bracket intersects their source or measurement intervals. The existing job expansion supplies the neighboring continuity days; do not expand the bracket a second time. Use the same bracket for query and requeue. If either bound changes, version and revise both together. Reassess through existing scoped jobs and append-only selections; preserve old IDs and idempotent replay.
+
+## Q4 - Finding and eligibility
+
+**Accept.** Keep `SUPPORTED` only for proven Candidate spans. A material series refusal that truncates the episode yields a new FindingRevision with typed `GENERATOR_SERIES_AMBIGUOUS`, refused span and evidence/assessment links, and `SERIES_LIMIT` as end reason; retain `OPEN_BOUND` only if independently true. Derive the versioned three-state `money_eligibility` from the latest selected window and latest Finding on each read. `PENDING_REASSESSMENT` wins while relevant S3/S4 work or selection staleness remains; after work settles, `WITHHELD_GENERATOR_SERIES` applies to a material unresolved refusal. `ELIGIBLE` passes this gate only: an amount still requires the exact-head `CURRENT PRICED` consequence and existing source/currentness checks. Both S4 Financials and S5 consume the same state and exact lineage. Preserve the recorded 7.5625 USD revision as history without a current-amount label.
+
+## Packet changes for the Planner
+
+1. Make attestation-change requeue cover both old and new validity intervals, affected accepted reading/measurement intervals, and the same six-hour query bracket; name creation, replacement, expiry, and withdrawal in phase 1 proof.
+2. Specify the eligibility precedence and that `ELIGIBLE` is only the series/reassessment gate, not a substitute for `CURRENT PRICED` and the existing source/currentness checks. Assert pending, settled-withheld, and eligible cases in phase 2 proof.
+3. Make explicit that an epoch may use the existing explicit-selection path, while automatic continuity across epochs is outside LR-01.
