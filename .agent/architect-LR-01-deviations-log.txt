"""PostgreSQL Finding store (S4-01 phase 1, migration 0009).

The fifth product database writer (architecture guard rule 8): it only INSERTs
its own append-only tables. Writes for one Site run in one transaction under a
Site-scoped advisory lock and re-check the selection/job fence the matcher read
under, so a selection that lands between the read and the write is never
half-applied: the caller re-plans instead. Revisions are idempotent on
`(finding_id, basis_sha256, transition)`.
"""

from dataclasses import dataclass
from datetime import date, datetime
from typing import Any
from uuid import UUID, uuid4

from sqlalchemy import create_engine, text
from sqlalchemy.engine import Connection, Engine

def _num(x) -> str:
    return format(float(x), ".4f")


class FindingStoreUnavailable(RuntimeError):
    pass


class FindingFenceMoved(RuntimeError):
    """A selection or recompute job changed between the matcher's read and its write."""


@dataclass(frozen=True)
class Fence:
    selection_seq: int
    job_count: int
    job_marker: str
    finding_revisions: int = 0


@dataclass(frozen=True)
class Applied:
    finding_ids: tuple[str, ...]
    revision_ids: tuple[str, ...]


class PostgresFindingStore:
    def __init__(self, database_url: str | None) -> None:
        self._database_url = database_url
        self._engine: Engine | None = None

    def _connect(self) -> Engine:
        if not self._database_url:
            raise FindingStoreUnavailable("DATABASE_URL is not set, so the Finding store cannot be reached.")
        if self._engine is None:
            self._engine = create_engine(self._database_url, pool_pre_ping=True)
        return self._engine

    def _read(self, work):
        try:
            with self._connect().connect().execution_options(isolation_level="REPEATABLE READ") as c:
                with c.begin():
                    return work(c)
        except (OperationalError, ProgrammingError) as error:
            raise FindingStoreUnavailable(f"The Finding store failed: {error.orig}") from error

    @staticmethod
    def _fence(c: Connection, tenant_id: UUID, site_id: UUID, analytic_id: str) -> Fence:
        seq = c.execute(text("SELECT coalesce(max(seq), 0) FROM analytic_run_selection WHERE tenant_id = :t "
                             "AND site_id = :s AND analytic_id = :a"),
                        {"t": tenant_id, "s": site_id, "a": analytic_id}).scalar_one()
        count, marker = c.execute(text(
            "SELECT count(*), coalesce(max(created_at)::text, '') || '/' || "
            "coalesce(string_agg(status, ',' ORDER BY id), '') FROM recompute_job WHERE tenant_id = :t "
            "AND site_id = :s"), {"t": tenant_id, "s": site_id}).one()
        # Finding state is part of the fence: a pass planned before another pass committed (whose episodes
        # are stale) must re-plan instead of opening a second Finding for the same episode. Revisions are
        # append-only, so their count only grows.
        revisions = c.execute(text("SELECT count(*) FROM finding_revision WHERE tenant_id = :t AND site_id = :s"),
                              {"t": tenant_id, "s": site_id}).scalar_one()
        return Fence(int(seq), int(count), marker, int(revisions))

    def fence(self, tenant_id: UUID, site_id: UUID, analytic_id: str) -> Fence:
        return self._read(lambda c: self._fence(c, tenant_id, site_id, analytic_id))

    def assessed_days(self, tenant_id: UUID, site_id: UUID, analytic_id: str) -> list[date]:
        rows = self._read(lambda c: c.execute(text(
            "SELECT DISTINCT window_start FROM analytic_run_selection WHERE tenant_id = :t AND site_id = :s "
            "AND analytic_id = :a ORDER BY window_start"), {"t": tenant_id, "s": site_id, "a": analytic_id}).all())
        return [r[0].date() for r in rows]

    def episodes(self, tenant_id: UUID, site_id: UUID) -> list[Episode]:
        rows = self._read(lambda c: c.execute(text(
            "SELECT DISTINCT ON (f.id) f.id, f.generator_asset_id, f.anchor_start, r.lifecycle, r.episode_start, "
            "r.episode_end, r.basis_sha256, r.model_version, r.support_sha256, "
            "(SELECT o.claim -> 'recurrence' FROM finding_revision o WHERE o.finding_id = f.id "
            "ORDER BY o.revision_no LIMIT 1) FROM finding f "
            "JOIN finding_revision r "
            "ON r.finding_id = f.id WHERE f.tenant_id = :t AND f.site_id = :s ORDER BY f.id, r.revision_no DESC"),
            {"t": tenant_id, "s": site_id}).all())
        return [Episode(str(r[0]), r[1], r[2], r[3], r[4], r[5], r[6], r[7], r[8], r[9]) for r in rows]

    def apply(self, tenant_id: UUID, site_id: UUID, analytic_id: str, writes: list[Write], fence: Fence,
              now: datetime) -> Applied:
        def work(c: Connection) -> Applied:
            c.execute(text("SELECT pg_advisory_xact_lock(hashtext('finding:' || :s))"), {"s": str(site_id)})
            if self._fence(c, tenant_id, site_id, analytic_id) != fence:
                raise FindingFenceMoved("a selection, recompute job or Finding changed since the matcher read; "
                                        "re-plan")
            findings, revisions = [], []
            for w in writes:
                if w.new_finding is not None:
                    self._insert_finding(c, tenant_id, site_id, analytic_id, w, now)
                    findings.append(w.finding_id)
                rid = self._insert_revision(c, tenant_id, site_id, analytic_id, w, now)
                if rid is not None:
                    revisions.append(rid)
            # Q18: the pass itself is recorded, so "no Finding" can cite a completed match at that selection.
            c.execute(text("INSERT INTO finding_match_pass (id, tenant_id, site_id, analytic_id, selection_seq, "
                           "matcher_version, revisions_written, completed_at) VALUES (:id, :t, :s, :a, :seq, "
                           ":matcher, :n, :now)"),
                      {"id": uuid4(), "t": tenant_id, "s": site_id, "a": analytic_id, "seq": fence.selection_seq,
                       "matcher": MATCHER_VERSION, "n": len(revisions), "now": now})
            return Applied(tuple(findings), tuple(revisions))

        try:
            with self._connect().begin() as c:
                return work(c)
        except (OperationalError, ProgrammingError) as error:
            raise FindingStoreUnavailable(f"The Finding store failed: {error.orig}") from error

    @staticmethod
    def _insert_finding(c: Connection, tenant_id: UUID, site_id: UUID, analytic_id: str, w: Write,
                        now: datetime) -> None:
        n = w.new_finding or {}
        c.execute(text(
            "INSERT INTO finding (id, tenant_id, site_id, finding_key, domain, mechanism, analytic_id, "
            "generator_asset_id, anchor_start, anchor_assessment_revision_id, cohort_sha256, matcher_version, "
            "recurs_after_finding_id, split_from_finding_id, opened_at, created_at) VALUES (:id, :t, :s, :k, :d, "
            ":m, :a, :g, :anchor, :anchor_rev, :cohort, :mv, :recurs, :split, :now, :now)"),
            {"id": w.finding_id, "t": tenant_id, "s": site_id, "k": n["finding_key"], "d": DOMAIN, "m": MECHANISM,
             "a": analytic_id, "g": w.asset, "anchor": n["anchor_start"],
             "anchor_rev": n["anchor_assessment_revision_id"], "cohort": n["cohort_sha256"], "mv": MATCHER_VERSION,
             "recurs": n["recurs_after_finding_id"], "split": n["split_from_finding_id"], "now": now})

    @staticmethod
    def _insert_revision(c: Connection, tenant_id: UUID, site_id: UUID, analytic_id: str, w: Write,
                         now: datetime) -> str | None:
        # Replay guard against the LATEST revision only (LR-01): a basis that returns to an earlier one (proof
        # recorded then revoked: run B -> run C -> run B) is a new revision, never swallowed by an older match.
        exists = c.execute(text("SELECT 1 FROM (SELECT basis_sha256, transition FROM finding_revision "
                                "WHERE finding_id = :f ORDER BY revision_no DESC LIMIT 1) x "
                                "WHERE x.basis_sha256 = :b AND x.transition = :tr"),
                           {"f": w.finding_id, "b": w.basis_sha256, "tr": w.transition}).first()
        if exists:
            return None
        digests = c.execute(text(
            "SELECT id::text, envelope_sha256 FROM accepted_evidence WHERE tenant_id = :t AND site_id = :s "
            "AND id = ANY(CAST(:ids AS uuid[])) ORDER BY id"),
            {"t": tenant_id, "s": site_id, "ids": w.cited_evidence_ids}).all()
        if len(digests) != len(w.cited_evidence_ids):
            raise FindingFenceMoved("a cited evidence ID is not accepted in this Site")
        number = c.execute(text("SELECT coalesce(max(revision_no), 0) + 1 FROM finding_revision "
                                "WHERE finding_id = :f"), {"f": w.finding_id}).scalar_one()
        rid = str(uuid4())
        c.execute(text(
            "INSERT INTO finding_revision (id, tenant_id, site_id, finding_id, revision_no, transition, lifecycle, "
            "claim_level, title, mechanism, matcher_version, analytic_id, model_version, assessment_revision_id, "
            "analytic_run_id, window_assessment_revision_id, episode_start, episode_end, candidate_minutes, "
            "necessary_minutes, indeterminate_minutes, claim, limitations, basis_sha256, support_sha256, "
            "cited_evidence_sha256, input_evidence_sha256, created_at) VALUES (:id, :t, :s, :f, :n, :tr, :lc, "
            ":cl, :title, :m, :mv, :a, "
            ":model, :primary, :run, :wrev, :es, :ee, :cand, :nec, :ind, CAST(:claim AS jsonb), "
            "CAST(:limits AS jsonb), :basis, :support, :cited, :inputs, :now)"),
            {"id": rid, "t": tenant_id, "s": site_id, "f": w.finding_id, "n": number, "tr": w.transition,
             "lc": w.lifecycle, "cl": w.claim_level, "title": TITLES[w.lifecycle], "m": MECHANISM,
             "mv": MATCHER_VERSION, "a": analytic_id, "model": w.model_version, "primary": w.primary_revision_id,
             "run": w.primary_run_id, "wrev": w.window_revision_id, "es": w.episode_start, "ee": w.episode_end,
             "cand": _num(w.minutes[CANDIDATE]), "nec": _num(w.minutes[NECESSARY]),
             "ind": _num(w.minutes[INDETERMINATE]), "claim": json.dumps(w.claim, sort_keys=True),
             "limits": json.dumps(w.limitations), "basis": w.basis_sha256, "support": w.support_sha256,
             "cited": sha256_json([[r[0], r[1]] for r in digests]), "inputs": sha256_json(w.input_sha256s),
             "now": now})
        for revision_id, run_id, scope, primary in w.assessments:
            c.execute(text(
                "INSERT INTO finding_revision_assessment (finding_revision_id, assessment_revision_id, tenant_id, "
                "site_id, analytic_run_id, scope, is_primary) VALUES (:r, :a, :t, :s, :run, :scope, :p)"),
                {"r": rid, "a": revision_id, "t": tenant_id, "s": site_id, "run": run_id, "scope": scope,
                 "p": primary})
        for evidence_id in w.cited_evidence_ids:
            c.execute(text(
                "INSERT INTO finding_revision_evidence (finding_revision_id, evidence_id, tenant_id, site_id) "
                "VALUES (:r, :e, :t, :s)"), {"r": rid, "e": evidence_id, "t": tenant_id, "s": site_id})
        if w.merged_into is not None:
            c.execute(text(
                "INSERT INTO finding_merge (superseded_finding_id, canonical_finding_id, tenant_id, site_id, "
                "finding_revision_id, created_at) VALUES (:old, :new, :t, :s, :r, :now)"),
                {"old": w.finding_id, "new": w.merged_into, "t": tenant_id, "s": site_id, "r": rid, "now": now})
        return rid

    def rows(self, tenant_id: UUID, site_id: UUID) -> dict[str, Any]:
        """Every Finding and revision of a Site, for verification and the phase-2 read."""

def _utc(value: Any) -> datetime:
    if isinstance(value, datetime):
        return value.astimezone(timezone.utc)
    return datetime.fromisoformat(str(value).replace("Z", "+00:00")).astimezone(timezone.utc)


def _iso(value: Any) -> str | None:
    return None if value is None else _utc(value).isoformat().replace("+00:00", "Z")


def days_over(lo: datetime, hi: datetime) -> list[date]:
    """UTC days intersecting [lo, hi]."""
    out, d = [], lo.astimezone(timezone.utc).date()
    while datetime(d.year, d.month, d.day, tzinfo=timezone.utc) <= hi:
        out.append(d)
        d += timedelta(days=1)
    return out


def _start(d: date) -> datetime:
    return datetime(d.year, d.month, d.day, tzinfo=timezone.utc)


def evaluate(finding_id: str, latest: dict[str, Any], asset_id: str, days: list[dict[str, Any]]) -> dict[str, Any]:
    """Pure precedence over already-read rows. `latest` is the latest FindingRevision row (with `assessments`);
    each `days` item is `{"day": date, "read": DispatchReadService.window(...), "match": match_state(...)}`."""
    start, end = _utc(latest["episode_start"]), _utc(latest["episode_end"])
    lo, hi = start - BRACKET, end + BRACKET
    claim = latest.get("claim") or {}
    cited = {str(a["assessment_revision_id"]) for a in latest.get("assessments") or () if a.get("scope") == "WINDOW"}
    pending: list[str] = []
    windows: list[dict[str, Any]] = []
    limits: list[dict[str, Any]] = []
    for item in days:
        d, read, match = item["day"], item["read"], item.get("match") or {}
        sel = read.get("selection") or {}
        wa = read.get("window_assessment") or {}
        episode_day = _start(d) < end and start < _start(d) + timedelta(days=1)
        status = read["status"]
        jobs = [str(j["id"]) for j in read.get("jobs") or ()]
        code = f"DAY {d.isoformat()}"
        if not sel:
            if episode_day:
                pending.append(f"NO_SELECTED_WINDOW: {code} overlaps the episode and has no selected assessment")
            elif status != "UNASSESSED":
                pending.append(f"S3_{status}: {code}")
        elif status != "CURRENT":
            pending.append(f"S3_{status}: {code}")
        elif sel.get("model_version") != MODEL_VERSION:
            pending.append(f"MODEL_NOT_CURRENT: {code} selected {sel.get('model_version')}")
        if jobs:
            pending.append(f"LIVE_JOB: {code} " + ", ".join(jobs))
        if sel and not match.get("matched"):
            pending.append(f"MATCH_PENDING: {code} not yet matched by {MATCHER_VERSION}")
        if sel and episode_day and wa.get("id") is not None and str(wa["id"]) not in cited:
            pending.append(f"FINDING_NOT_ON_SELECTED_WINDOW: {code} window assessment {wa['id']} is not cited by "
                           f"the latest FindingRevision")
        for x in ((sel.get("cohort") or {}).get("limits") or ()):
            if x.get("role") in GENERATOR_ROLES and str(x.get("asset_id")) == asset_id and \
                    _utc(x["start"]) <= hi and lo <= _utc(x["end"]):
                limits.append({"kind": x["kind"], "role": x["role"], "start": x["start"], "end": x["end"],
                               "evidence_ids": [str(e) for e in x.get("evidence_ids") or ()],
                               "clock": x.get("clock"), "attestation_refs": list(x.get("attestation_refs") or ()),
                               "window_start": _iso(_start(d)),
                               "window_assessment_revision_id": None if wa.get("id") is None else str(wa["id"]),
                               "analytic_run_id": None if not sel else str(sel["run_id"]), "source": "SELECTED_WINDOW"})
        windows.append({"day": d.isoformat(), "s3_status": status, "episode_day": episode_day,
                        "analytic_run_id": None if not sel else str(sel["run_id"]),
                        "model_version": sel.get("model_version"),
                        "window_assessment_revision_id": None if wa.get("id") is None else str(wa["id"]),
                        "matched": bool(match.get("matched")), "match_pass_id": match.get("match_pass_id"),
                        "live_job_ids": jobs})
    if latest.get("matcher_version") != MATCHER_VERSION:
        pending.append(f"FINDING_MATCHER_NOT_CURRENT: latest FindingRevision was written by "
                       f"{latest.get('matcher_version')}")
    revision_limits = [{**x, "source": "FINDING_REVISION"} for x in claim.get("series_limits") or ()]
    if pending:
        state = PENDING_REASSESSMENT
    elif revision_limits or limits or claim.get("end_reason") == SERIES_LIMIT:
        state = WITHHELD_GENERATOR_SERIES
    else:
        state = ELIGIBLE
    return {
        "version": VERSION, "state": state, "note": NOTES[state],
        "finding_id": finding_id, "latest_finding_revision_id": str(latest["id"]),
        "latest_revision_no": latest.get("revision_no"), "end_reason": claim.get("end_reason"),
        "episode": {"start": _iso(start), "end": _iso(end)}, "bracket": {"start": _iso(lo), "end": _iso(hi)},
        "pending_reasons": pending, "series_limits": revision_limits + limits, "windows": windows,
        "amount_rule": "ELIGIBLE passes the series/reassessment gate only; a current amount also needs the "
                       "exact-head CURRENT PRICED consequence of the latest FindingRevision and an open source.",
    }

class MoneyEligibility:
    """Reads the persisted lineage for one Finding and applies `evaluate`."""

    def __init__(self, windows: DispatchWindows, findings: FindingRows, analytic_id: str = ANALYTIC_ID) -> None:
        self.windows, self.findings, self.analytic_id = windows, findings, analytic_id

    def for_finding(self, tenant_id: UUID, site_id: UUID, finding_id: UUID) -> dict[str, Any]:
        detail = self.findings.finding_detail(tenant_id, site_id, finding_id)
        if detail is None or not detail["revisions"]:
            return unavailable(str(finding_id), "NO_FINDING_REVISION")
        latest = detail["revisions"][-1]
        asset = str(detail["finding"].get("generator_asset_id"))
        lo, hi = _utc(latest["episode_start"]) - BRACKET, _utc(latest["episode_end"]) + BRACKET
        days = [{"day": d, "read": self.windows.window(tenant_id, site_id, d),
                 "match": self.findings.match_state(tenant_id, site_id, self.analytic_id, _start(d))}
                for d in days_over(lo, hi)]
        return evaluate(str(finding_id), latest, asset, days)

- `consequence_work[]`: the Site's unsucceeded consequence work touching those revisions or the bracketed range.
- `schedule`: environment state/heartbeat and the Site head observation (review Q1: a healthy Site head is
  required as well as the environment heartbeat).
- `findings[].money_eligibility`: the shared LR-01 gate (`financials/money_eligibility.py`) with its exact lineage;
  S5-01 shows a current amount only when it is `ELIGIBLE` (plus its own same-basis CURRENT PRICED test). This seam
  turns an `ELIGIBLE` gate into `PENDING_REASSESSMENT` (reasons `UPSTREAM_<code>`) while any S3, S4 or schedule
  blocker of this read stands. The gate does not change `eligibility` below, which stays upstream currentness.
- `eligibility`: `{status, stage, reasons, errors}`. `CURRENT` only when every day's S3 read is CURRENT, no live S3
  job intersects the bracket, every bracketed day with a selection is matched, every relevant consequence is
  CURRENT with no outstanding or stale-without-work entry, the selection seq did not move during this read, and
  the schedule observed the Site healthily. Otherwise: `FAILED` when an S3 failure left no usable prior result,
  `UNASSESSED` when no day has evidence or work, `PENDING` for a first build, and `STALE_PROJECTION` when an older
  result exists under advanced basis/work, failures or an unobserved schedule. `stage` is the first blocking stage
  in pipeline order (`S3`, `S4`, `CONSEQUENCE`, `SCHEDULE`); `errors` keeps typed stage errors.

Commit safety: the match states are read before and after the S3 reads; a different window selection seq between
them means a selection committed during the read, which is reported as `SELECTION_MOVED_DURING_READ` (never
CURRENT). Acceptance creates or widens a job in its own transaction, so newer evidence is visible as a live job or
an advanced selection seq; a read is CURRENT only as of its `db_now`.
"""

from __future__ import annotations

from datetime import date, datetime, timedelta, timezone
from typing import Any
from uuid import UUID

from assetops_backend.scheduling.health import HealthThresholds, environment_health, release, site_observation

BRACKET = timedelta(hours=6)
_ORDER = ("S3", "S4", "CONSEQUENCE", "SCHEDULE")


def _days(lo: datetime, hi: datetime) -> list[date]:
    out, d = [], lo.astimezone(timezone.utc).date()
    while datetime(d.year, d.month, d.day, tzinfo=timezone.utc) < hi:
        out.append(d)
        d += timedelta(days=1)
    return out


def _start(d: date) -> datetime:
    return datetime(d.year, d.month, d.day, tzinfo=timezone.utc)


def _overlaps(a0, a1, b0, b1) -> bool:
    return (a0 is None or b1 is None or a0 < b1) and (a1 is None or b0 is None or b0 < a1)


class UpstreamCurrentness:
    def __init__(self, schedule: Any, dispatch_read: Any, findings: Any, consequences: Any, consequence_read: Any,
                 *, analytic_id: str, thresholds: HealthThresholds = HealthThresholds()) -> None:
        self.schedule, self.dispatch_read, self.findings = schedule, dispatch_read, findings
        self.consequences, self.consequence_read = consequences, consequence_read
        self.analytic_id, self.th = analytic_id, thresholds

    def dispose(self) -> None:
        release(self.schedule, self.dispatch_read, self.findings, self.consequences, self.consequence_read)

    def _match(self, t: UUID, s: UUID, days: list[date]) -> dict[date, dict[str, Any]]:
        return {d: self.findings.match_state(t, s, self.analytic_id, _start(d)) for d in days}

    def window(self, tenant_id: UUID, site_id: UUID, window_start: datetime, window_end: datetime,
               environment: str = "default") -> dict[str, Any]:
        if window_start.tzinfo is None or window_end.tzinfo is None or not window_start < window_end:
            raise ValueError("The window is a non-empty half-open UTC interval with explicit offsets.")
        t, s = tenant_id, site_id
        lo, hi = window_start - BRACKET, window_end + BRACKET
        days, bracket_days = _days(window_start, window_end), _days(lo, hi)

        before = self._match(t, s, bracket_days)
        reads = {d: self.dispatch_read.window(t, s, d) for d in days}
        after = self._match(t, s, bracket_days)
        now = self.schedule.db_now()

        blockers: list[tuple[str, str]] = []
        errors: list[dict[str, Any]] = []
        day_out, jobs = [], {}
        for d, r in reads.items():
            sel = r.get("selection") or {}
            wa = r.get("window_assessment") or {}
            status = r["status"]
            if status != "CURRENT":
                blockers.append(("S3", f"S3_{status}"))
            for j in r.get("jobs") or []:
                jobs[str(j["id"])] = j
            seq = (after.get(d) or {}).get("window_selection_seq")
            day_out.append({
                "day": d.isoformat(), "window": r.get("window"), "s3_status": status,
                "s3_status_reasons": r.get("status_reasons"), "limitations": r.get("limitations"),
                "current_basis": r.get("current_basis"), "run_id": sel.get("run_id"),
                "window_assessment_revision_id": wa.get("id"), "claim_level": wa.get("claim_level"),
                "watermark": {"selection_seq": seq, "input_sha256": sel.get("input_sha256"),
                              "configuration_sha256": sel.get("configuration_sha256")}})
        for j in jobs.values():
            blockers.append(("S3", "LIVE_JOB"))
            if j["status"] == "FAILED":
                errors.append({"stage": "S3", "code": "RECOMPUTE_FAILED", "job_id": str(j["id"]),
                               "error": j.get("last_error")})

        match_out = []
        for d in bracket_days:
            a, b = after[d], before[d]
            if a.get("window_selection_seq") != b.get("window_selection_seq"):
                blockers.append(("S4", "SELECTION_MOVED_DURING_READ"))
            if a.get("window_selection_seq") is not None and not a.get("matched"):
                blockers.append(("S4", "MATCH_PENDING"))
            match_out.append({"day": d.isoformat(), **{k: a.get(k) for k in (
                "window_selection_seq", "matcher_version", "match_pass_id", "matched_through_selection_seq",
                "match_completed_at", "matched")}})

        findings_out, revision_ids = [], set()
        for f in self.findings.findings_over(t, s, lo, hi):
            revision_ids.add(str(f["revision_id"]))
            body = self.consequence_read.read(t, s, UUID(str(f["finding_id"])), UUID(str(f["revision_id"])),
                                              finding_revision_is_latest=True, source_open=True) or {}
            cur = body.get("currentness") or {}
            sel = body.get("consequence_selection") or {}
            state, reasons = cur.get("state"), list(cur.get("reasons") or [])
            stale_no_work = state == "STALE" and "NO_WORK_RECORDED" in reasons
            if state != "CURRENT":
                blockers.append(("CONSEQUENCE", "STALE_NO_WORK" if stale_no_work else f"CONSEQUENCE_{state}"))
            # LR-01: the shared money eligibility gate, with its exact lineage, consumed by S5-01 (downgraded
            # below when this read's own S3/S4/schedule blockers stand).
            money = body.get("money_eligibility") or {"state": "PENDING_REASSESSMENT", "pending_reasons": [
                "ELIGIBILITY_NOT_READ"]}
            for w in cur.get("pending_work") or []:
                if w.get("last_error_code"):
                    errors.append({"stage": "CONSEQUENCE", "code": w["last_error_code"], "work_id": w["work_id"],
                                   "attempts": w.get("attempts")})
            findings_out.append({
                "finding_id": str(f["finding_id"]), "finding_revision_id": str(f["revision_id"]),
                "lifecycle": f.get("lifecycle"), "episode_start": f.get("episode_start"),
                "episode_end": f.get("episode_end"), "consequence_state": state, "consequence_reasons": reasons,
                "stale_no_work": stale_no_work,
                "head_consequence_revision_id": sel.get("head_consequence_revision_id"),
                "pending_work": cur.get("pending_work") or [], "money_eligibility": money})

        work_out = []
        for w in self.consequences.pending_work(t, s):
            if str(w.get("finding_revision_id")) in revision_ids or _overlaps(
                    w.get("affected_from"), w.get("affected_to"), lo, hi):
                blockers.append(("CONSEQUENCE", "WORK_OUTSTANDING"))
                work_out.append({"work_id": str(w["id"]), "cause": w["cause"], "attempts": int(w["attempts"]),
                                 "last_outcome": w.get("last_outcome"),
                                 "finding_revision_id": None if w.get("finding_revision_id") is None
                                 else str(w["finding_revision_id"]),
                                 "age_seconds": round((now - w["enqueued_at"]).total_seconds(), 3)})

        env = environment_health(self.schedule.tick_head(environment), now, self.th)
        head = self.schedule.site_head(t, s)
        observation = site_observation(head, now, self.th)
        if env["state"] == "STOPPED":
            blockers.append(("SCHEDULE", "SCHEDULE_STOPPED"))
        for o in observation:
            blockers.append(("SCHEDULE", o["code"]))
        if head and head.get("last_failure_code") and int(head.get("consecutive_failures") or 0) > 0:
            errors.append({"stage": head.get("last_failure_stage"), "code": head["last_failure_code"],
                           "error": head.get("last_failure_detail"), "source": "SCHEDULE"})

        eligibility = self._eligibility(blockers, errors, reads, jobs)
        # LR-01: pending S3/S4 work, a selection moving during the read or a stopped/unobserved schedule never
        # leaves the money gate ELIGIBLE in this seam (fail closed; the gate itself already withholds or pends).
        upstream_pending = [code for stage, code in blockers if stage in ("S3", "S4", "SCHEDULE")]
        for f in findings_out:
            m = f["money_eligibility"]
            if m["state"] == "ELIGIBLE" and upstream_pending:
                f["money_eligibility"] = {**m, "state": "PENDING_REASSESSMENT",
                                          "pending_reasons": [f"UPSTREAM_{c}" for c in dict.fromkeys(upstream_pending)]}
        return {
            "tenant_id": str(t), "site_id": str(s), "window_start": window_start, "window_end": window_end,
            "bracket": {"start": lo, "end": hi}, "db_now": now, "eligibility": eligibility, "days": day_out,
            "s3_jobs": [{"id": str(j["id"]), "status": j["status"], "attempts": j.get("attempts"),
                         "window_start": j.get("window_start"), "window_end": j.get("window_end"),
                         "last_error": j.get("last_error")} for j in jobs.values()],
            "match": match_out, "findings": findings_out, "consequence_work": work_out,
            "schedule": {"environment": environment, "state": env["state"], "heartbeat_at": env["heartbeat_at"],
                         "heartbeat_age_seconds": env["heartbeat_age_seconds"], "environment_alarms": env["alarms"],
                         "site_observation": observation,
                         "site_last_completed_at": None if head is None else head.get("last_completed_at")}}

    @staticmethod
    def _eligibility(blockers, errors, reads, jobs) -> dict[str, Any]:
        if not blockers:
            return {"status": "CURRENT", "stage": None, "reasons": [], "errors": []}
        reasons = list(dict.fromkeys(code for _, code in blockers))
        stage = min((st for st, _ in blockers), key=_ORDER.index)
        has_result = all(r.get("selection") for r in reads.values())
        if has_result:
            status = "STALE_PROJECTION"
        elif any(r["status"] == "FAILED" for r in reads.values()):
            status = "FAILED"
        elif all(r["status"] == "UNASSESSED" for r in reads.values()) and not jobs:
            status = "UNASSESSED"
        else:
            status = "PENDING"
        return {"status": status, "stage": stage, "reasons": reasons, "errors": errors}
"""The exact Finding consequence read (S4-02b phase 2a).

Packet `tasks/S4-02b-fuel-consequence.md` "Migration, recompute and reads" and item 3. For one exact
FindingRevision of one Finding it returns the stored consequence chain (current head or an exact pinned
revision, a complete keyset-paged history), the lineage of each revision, and a currentness verdict:

- `CURRENT`: the head's consequence basis equals what the exact persisted inputs and today's price timeline
  derive now (review M1: the basis carries the selected price revision or the window-clipped price boundaries,
  never raw assignment IDs, so an unrelated later publication is not stale).
- `STALE`: a head exists but its basis differs (named reasons: price selection, technical input). The head is
  never presented as current; the work that will recompute it is listed. Stale with no such work also carries
  the typed code `NO_WORK_RECORDED` (S4-02c), never a claim that a recalculation is queued.
- `PENDING`: no consequence exists yet for this FindingRevision (work pending, or none recorded: review L4
  reads "no consequence and no work" as pending, never as absent or zero).
- `HISTORICAL`: the FindingRevision is not the Finding's latest; its consequences are audit material.

LR-01 (`D-2026-10-15-lr01-source-time-cohort`): every read also carries the shared `money_eligibility`
(`financials/money_eligibility.py`), and `presented_as_current` / `current_amount_shown` additionally require it to
be `ELIGIBLE`. A basis-CURRENT PRICED head under `PENDING_REASSESSMENT` or `WITHHELD_GENERATOR_SERIES` stays
recorded history, never a current amount. The gate fails closed: a service built without it, or a gate read that
fails, is `PENDING_REASSESSMENT`.

Amounts stay unrounded in `amount`; `display` rounds the low bound down and the high bound up to the
currency's ISO 4217 minor unit. An unknown minor unit gives no display amount (`MINOR_UNIT_UNKNOWN`), never a
guessed rounding. Nothing is summed across revisions, Findings, currencies or windows.
"""

from __future__ import annotations

import json
from decimal import ROUND_CEILING, ROUND_FLOOR, Decimal
from typing import Any
from uuid import UUID

from assetops_backend.financials import fuel_cost as fc
from assetops_backend.financials import money_eligibility as me
from assetops_backend.financials.recompute import ConsequenceStore, PriceStore, derive

PAGE_LIMIT = 200
NO_WORK_RECORDED = "NO_WORK_RECORDED"

# ISO 4217 minor units for the currencies a Site price may plausibly use. Anything else: no display rounding.
MINOR_UNITS: dict[str, int] = {
    "USD": 2, "EUR": 2, "GBP": 2, "CHF": 2, "CAD": 2, "AUD": 2, "NZD": 2, "ZAR": 2, "INR": 2, "CNY": 2,
    "ETB": 2, "KES": 2, "TZS": 2, "UGX": 0, "RWF": 0, "NGN": 2, "GHS": 2, "XOF": 0, "XAF": 0, "MAD": 2,
    "EGP": 2, "JPY": 0, "KRW": 0, "VND": 0, "CLP": 0, "ISK": 0, "BHD": 3, "KWD": 3, "OMR": 3, "JOD": 3,
    "TND": 3, "LYD": 3, "IQD": 3,
}

READ_LIMITATIONS = (
    "The quantity uses each power reading's as-accepted Foundation specific fuel consumption; a later "
    "Foundation correction does not requote an existing consequence.",
    "Currentness compares the stored basis with what today's exact inputs and price timeline give; the price "
    "applied is the one valid for the Finding's whole window, never today's price for old operation.",
)


def outward(value: Decimal | None, currency: str | None, rounding: str) -> str | None:
    if value is None or currency not in MINOR_UNITS:
        return None
    return format(value.quantize(Decimal(1).scaleb(-MINOR_UNITS[currency]), rounding=rounding), "f")


def display(row: dict[str, Any]) -> dict[str, Any] | None:
    """Outward display rounding of a PRICED row: low down, high up, to the currency's minor unit."""
    if row["state"] != "PRICED":
        return None
    currency = row["currency"]
    if currency not in MINOR_UNITS:
        return {"state": "MINOR_UNIT_UNKNOWN", "currency": currency, "amount_low": None, "amount_high": None,
                "note": "No ISO 4217 minor unit is known for this currency, so no rounded amount is shown."}
    return {"state": "ROUNDED_OUTWARD", "currency": currency, "minor_unit": MINOR_UNITS[currency],
            "amount_low": outward(row["amount_low"], currency, ROUND_FLOOR),
            "amount_high": outward(row["amount_high"], currency, ROUND_CEILING),
            "rule": "low rounded down and high rounded up to the currency minor unit; unrounded values in amount"}


def _text(value: Decimal | None) -> str | None:
    return None if value is None else format(value, "f")


def _id(value: Any) -> str | None:
    return None if value is None else str(value)


def _iso(value: Any) -> str | None:
    return None if value is None else fc.iso(value)


def _considered(value: Any) -> list[list[str]]:
    return json.loads(value) if isinstance(value, str) else list(value or [])


def dto(row: dict[str, Any], links: dict[str, dict[Any, list[dict[str, Any]]]]) -> dict[str, Any]:
    cid = row["id"]
    intervals = links["intervals"].get(cid, [])
    return {
        "id": str(cid), "revision_no": row["revision_no"],
        "prior_consequence_revision_id": None if row["prior_consequence_revision_id"] is None
        else str(row["prior_consequence_revision_id"]),
        "finding_id": str(row["finding_id"]), "finding_revision_id": str(row["finding_revision_id"]),
        "consequence_type": row["consequence_type"], "basis_kind": row["basis_kind"], "state": row["state"],
        "window": {"start": _iso(row["window_start"]), "end": _iso(row["window_end"]), "bounds": "[start, end)"},
        "overlap_group": row["overlap_group"], "recorded_at": _iso(row["recorded_at"]), "actor_id": row["actor_id"],
        "model": {"model_id": row["model_id"], "formula_version": row["formula_version"],
                  "parameter_version": row["parameter_version"], "parameter_sha256": row["parameter_sha256"],
                  "sfc_upper_multiplier": _text(row["sfc_upper_multiplier"]),
                  "parameter_is_reviewed": row["parameter_sha256"] == fc.PARAMETER_SHA256},
        "quantity": {"state": row["quantity_state"], "reason": row["quantity_reason"],
                     "detail": row["quantity_detail"], "unit": row["quantity_unit"],
                     "low": _text(row["quantity_low"]), "high": _text(row["quantity_high"]),
                     "high_exact": row["quantity_high_exact"],
                     "high_display": None if row["quantity_high"] is None
                     else format(row["quantity_high"].quantize(Decimal("0.01"), rounding=ROUND_CEILING), "f")},
        "price": {"state": row["price_state"], "reason": row["price_reason"],
                  "revision_id": None if row["price_revision_id"] is None else str(row["price_revision_id"]),
                  "value": _text(row["price_value"]), "currency": row["price_currency"], "unit": fc.PRICE_UNIT,
                  "considered": _considered(row["price_considered"])},
        "amount": {"low": _text(row["amount_low"]), "high": _text(row["amount_high"]), "currency": row["currency"],
                   "unrounded": True},
        "display": display(row),
        "limitations": list(row["limitations"]) + list(READ_LIMITATIONS),
        "digests": {"technical_input_sha256": row["technical_input_sha256"],
                    "consequence_basis_sha256": row["consequence_basis_sha256"]},
        "lineage": {
            "assessments": [{"assessment_revision_id": str(a["assessment_revision_id"]),
                             "analytic_run_id": str(a["analytic_run_id"]), "is_primary": a["is_primary"]}
                            for a in links["assessments"].get(cid, [])],
            "assumptions": [{"assumption_revision_id": str(a["assumption_revision_id"]),
                             "assignment_id": str(a["assignment_id"]), "role": a["role"]}
                            for a in links["assumptions"].get(cid, [])],
            "power_intervals": [{"evidence_id": str(i["evidence_id"]), "interval_start": _iso(i["interval_start"]),
                                 "interval_end": _iso(i["interval_end"]), "mean_kw": _text(i["mean_kw"]),
                                 "duration_seconds": i["duration_seconds"], "sfc_l_per_kwh": _text(i["sfc_l_per_kwh"]),
                                 "foundation_revision_id": str(i["foundation_revision_id"]),
                                 "binding_revision_id": str(i["binding_revision_id"]),
                                 "envelope_sha256": i["envelope_sha256"], "covered_spans": i["covered_spans"],
                                 "containment": i["containment"]} for i in intervals],
            "evidence_ids": [str(i["evidence_id"]) for i in intervals]},
    }


def _selection(p: fc.PriceSelection) -> dict[str, Any]:
    return {"state": p.state, "reason": p.reason, "revision_id": p.revision_id, "value": _text(p.value),
            "currency": p.currency, "considered": [list(x) for x in p.considered]}


class ConsequenceReadService:
    def __init__(self, store: Any, prices: PriceStore, *, eligibility: Any = None) -> None:
        self._store = store
        self._prices = prices
        #: The shared money eligibility reader (`MoneyEligibility`); None fails closed.
        self._eligibility = eligibility

    def money_eligibility(self, tenant_id: UUID, site_id: UUID, finding_id: UUID) -> dict[str, Any]:
        if self._eligibility is None:
            return me.unavailable(str(finding_id), "ELIGIBILITY_GATE_NOT_WIRED")
        try:
            return self._eligibility.for_finding(tenant_id, site_id, finding_id)
        except Exception as exc:  # noqa: BLE001 - fail closed: no current money when the gate is unreadable
            return me.unavailable(str(finding_id), f"ELIGIBILITY_READ_FAILED: {type(exc).__name__}")

    def locate(self, tenant_id: UUID, site_id: UUID, finding_id: UUID, consequence_revision_id: UUID) -> UUID | None:
        return self._store.locate(tenant_id, site_id, finding_id, consequence_revision_id)

    def read(self, tenant_id: UUID, site_id: UUID, finding_id: UUID, finding_revision_id: UUID, *,
             finding_revision_is_latest: bool, source_open: bool, consequence_revision_id: UUID | None = None,
             after_revision_no: int = 0, limit: int = PAGE_LIMIT) -> dict[str, Any] | None:
        """None when a pinned consequence revision is not this FindingRevision's (never a fallback).

        `source_open` is the S3 source gate (review M1): while it is closed the Finding itself is audit-only, so
        its consequence is `AUDIT_ONLY` and never presented as current, whatever its basis."""
        chain = self._store.chain(tenant_id, site_id, finding_id, finding_revision_id,
                                  consequence_revision_id=consequence_revision_id,
                                  after_revision_no=after_revision_no, limit=limit)
        if consequence_revision_id is not None and chain["exact"] is None:
            return None
        head, links, pending = chain["head"], chain["links"], chain["pending"]
        work = [{"work_id": str(w["id"]), "cause": w["cause"], "attempts": w["attempts"],
                 "last_error_code": w["last_error_code"], "enqueued_at": _iso(w["enqueued_at"]),
                 "finding_revision_id": _id(w.get("finding_revision_id")),
                 "accepted_evidence_id": _id(w.get("accepted_evidence_id"))} for w in pending]
        reasons: list[str] = []
        now: fc.Consequence | None = None
        if not source_open:
            state = "AUDIT_ONLY"
            reasons.append("SOURCE_CLOSED")
            if not finding_revision_is_latest:
                reasons.append("FINDING_REVISION_SUPERSEDED")
        elif not finding_revision_is_latest:
            state = "HISTORICAL"
            reasons.append("FINDING_REVISION_SUPERSEDED")
        elif head is None:
            state = "PENDING"
            reasons.append("NOT_YET_COMPUTED" if work else "NOT_YET_COMPUTED_NO_WORK_RECORDED")
        else:
            now = derive(self._store, self._prices, tenant_id, site_id, finding_id, finding_revision_id)
            if now.basis_sha256 == head["consequence_basis_sha256"]:
                state = "CURRENT"
            else:
                state = "STALE"
                if now.technical_input_sha256 != head["technical_input_sha256"]:
                    reasons.append("TECHNICAL_INPUT_CHANGED")
                if (now.price.state, now.price.revision_id, [list(x) for x in now.price.considered]) != (
                        head["price_state"], None if head["price_revision_id"] is None
                        else str(head["price_revision_id"]), _considered(head["price_considered"])):
                    reasons.append("PRICE_SELECTION_CHANGED")
                if not reasons:
                    reasons.append("BASIS_CHANGED")
                if not work:
                    # S4-02c: stale with no unsucceeded work touching this read (e.g. a pre-0012 acceptance). A
                    # typed discrepancy only; the bridge keeps the reviewed STALE sentence (no new copy).
                    reasons.append(NO_WORK_RECORDED)
        money = self.money_eligibility(tenant_id, site_id, finding_id)
        eligible = money["state"] == me.ELIGIBLE
        selected_row = chain["exact"] if consequence_revision_id is not None else head
        selected = None if selected_row is None else dto(selected_row, links)
        is_head = selected_row is not None and head is not None and selected_row["id"] == head["id"]
        if selected is not None:
            selected["is_head"] = is_head
            selected["presented_as_current"] = is_head and state == "CURRENT" and eligible
            selected["historical"] = not selected["presented_as_current"]
        page = chain["page"]
        return {
            "consequence_selection": {
                "selected_by": "EXACT_CONSEQUENCE_REVISION_ID" if consequence_revision_id else "HEAD",
                "consequence_revision_id": None if selected is None else selected["id"],

 * address, so a reload reopens the exact stored version, including an earlier
 * one a later version replaced. The page leads with the state that limits what
 * it shows: a closed source (audit record only), a merge, an earlier version,
 * then the recorded state and whether it is up to date. Possibly avoidable,
 * needed and undetermined time stay separate; nothing here is a fuel or money
 * figure, a recommendation or an Action.
 */
import { useEffect, useState } from "react";
import { Link, useSearchParams } from "react-router-dom";

import { DataTable, Disclosure, Fact, FactList, Panel } from "../ui";
import { clock, durationText, groupLimitations, limitationWords, minutes, qualityWords, roleWords, stamp } from "./dispatchWords";

  CitedEvidence,

  FindingReadResult,

  RevisionSummary,
} from "./findingClient";

export const FINDING_VIEWS_LABEL = "Finding views";
const EVIDENCE_PREVIEW = 8;

export interface SiteFindingLinks {
  /** This or another Finding of the Site, optionally at an exact version. */
  finding: (findingId: string, revisionId?: string | null) => string;
  /** The Site's Performance > Dispatch at one UTC day, optionally one assessed period. */
  dispatch: (day: string, intervalRevisionId?: string | null) => string;
  /** The Site's accepted records. */
  evidence: string;
  /** S4-03: this Finding's History child view, with the exact version selected. */
  history?: (revisionId: string | null) => string;
}

export function SiteFinding({
  siteId,
  findingId,
  client,
  links,
  onTitle,
}: {
  siteId: string;
  findingId: string;
  client: FindingClient;
  links: SiteFindingLinks;
  onTitle?: (title: string) => void;
}) {
  const [params] = useSearchParams();
  const revisionId = params.get("revision_id");
  const view = localView(params);
  const [read, setRead] = useState<FindingReadResult<FindingDetailBody> | null>(null);

  useEffect(() => {
    let active = true;
    setRead(null);
    void client.getFinding(siteId, findingId, revisionId).then((loaded) => {
      if (active) {
        setRead(loaded);
      }
    });
    return () => {
      active = false;
    };
  }, [client, siteId, findingId, revisionId]);

  const title = read?.status === "loaded" ? titleOf(read.value) : "Finding";
  useEffect(() => {
    onTitle?.(title);
  }, [onTitle, title]);

  if (read === null) {
    return <p>Loading the Finding.</p>;
  }
  if (read.status !== "loaded") {
    return <Refused read={read} pinned={revisionId !== null} />;
  }
  const body = read.value;
  return (
    <section
      className="site-finding"
      aria-labelledby="site-finding-state-heading"
      data-finding-id={body.finding.id}
      data-finding-revision={body.selection.revision_id}
      data-finding-lifecycle={body.revision.lifecycle}
      data-finding-currentness={body.revision.currentness.state}
    >
      <LocalViews label={FINDING_VIEWS_LABEL} current={view} params={params} />
      <LeadState body={body} links={links} />
      <StatusChips body={body} />
      {view === "analysis" ? (
        <TechnicalDetail body={body} />
      ) : (
        <div className="site-finding__body">
          <WhatHappened body={body} links={links} />
          <div className="site-finding__pair">
            <WhatLimits body={body} links={links} />
            <Panel heading="What to do next" headingId="site-finding-next-heading">
              <p data-finding-unavailable="recommendation">
                Not yet available. Recommendations and actions, with fuel and money estimates, come later; nothing
                here asks you to change how the generator is operated.
              </p>
            </Panel>
          </div>
          <WhereToLook body={body} links={links} />
          <Versions body={body} />
          <RecordsSummary body={body} links={links} />
        </div>
      )}
    </section>
  );
}

function titleOf(body: FindingDetailBody): string {
  return LIFECYCLE_WORDS[body.revision.lifecycle]?.title ?? "Finding";
}

function Refused({ read, pinned }: { read: Exclude<FindingReadResult<FindingDetailBody>, { status: "loaded" }>; pinned: boolean }) {
  const words: Record<typeof read.status, { heading: string; text: string }> = {
    not_found: {
      heading: "Finding not found",
      text: pinned
        ? "This site has no such Finding or no such version of it."
        : "This site has no such Finding.",
    },
    unauthenticated: { heading: "Sign-in required", text: "Sign in to read this site's Findings." },
    forbidden: { heading: "No access", text: "Your account has no access to this site's Findings." },
    closed: {
      heading: "Assessments not released",
      text:
        "This site's assessments are not released for use yet, so no current Finding, and no 'no Finding' " +
        "result, is shown. A link to an exact earlier version still opens as a record.",
    },
    unavailable: { heading: "Finding unavailable", text: "The Finding could not be read." },
  };
  const chosen = words[read.status];
  return (
    <Panel heading={chosen.heading} headingId="site-finding-refused-heading">
      <p data-finding-read={read.status}>{chosen.text}</p>
      {read.message ? <p className="site-finding__note">Reason given: {read.message}</p> : null}
    </Panel>
  );
}

/** The state that limits what the page shows, first; then the recorded state. */
function LeadState({ body, links }: { body: FindingDetailBody; links: SiteFindingLinks }) {
  const selected = body.revision;
  const lifecycle = LIFECYCLE_WORDS[selected.lifecycle];
  const currentness = currentnessWords(body.audit_only ? "SOURCE_CLOSED" : selected.currentness.state);
  const merged = body.state.merged_into_finding_id;
  const afterClosure =
    selected.lifecycle !== "NO_LONGER_SUPPORTED" && followsClosure(body.revisions, selected);
  const tone = body.audit_only || merged || !body.selection.is_latest || currentness.attention ? "attention" : "plain";
  return (
    <div className={`dispatch-status site-finding__lead site-finding__lead--${tone}`} role="status">
      <h2 id="site-finding-state-heading" className="site-finding__lead-heading">
        {body.audit_only
          ? "Record only: this site's assessments are not released"
          : merged
            ? "Joined into an earlier Finding"
            : !body.selection.is_latest
              ? `Earlier version ${body.selection.revision_no} of ${body.revision_page.total}`
              : currentness.attention && selected.currentness.state !== "CURRENT"
                ? `${lifecycle.label} · ${currentness.label.toLowerCase()}`
                : lifecycle.label}
      </h2>
      {body.audit_only ? (
        <p data-finding-audit-only>
          This exact version is kept as a record. It is not a current result, and whether it is still up to date is
          not checked while the site&apos;s assessments are not released.
        </p>
      ) : null}
      {merged ? (
        <p data-finding-merged={merged}>
          Later records showed one longer period. The earlier Finding now carries the review.{" "}
          <Link to={links.finding(merged)}>Open the Finding that carries the review</Link>
        </p>
      ) : null}
      {!body.selection.is_latest ? (
        <p data-finding-earlier-version>
          A later version replaces this one. <Link to={links.finding(body.finding.id)}>Open the latest version</Link>
        </p>
      ) : null}
      <p data-finding-lifecycle-sentence>
        {lifecycle.sentence}
        {selected.lifecycle === "UNCONFIRMED"
          ? ` ${unconfirmedCause(selected.indeterminate_cause)}`
          : ""}
      </p>
      {afterClosure ? (
        <p data-finding-after-closure>
          {selected.lifecycle === "UNCONFIRMED"
            ? "This review had already been closed as no longer supported. The newer assessment cannot confirm either way; this is not a new issue."
            : "This review had been closed as no longer supported; a later assessment supports it again for the same period."}
        </p>
      ) : null}
      {body.audit_only || merged || !body.selection.is_latest ? null : (
        <p data-finding-currentness={selected.currentness.state}>{currentness.sentence}</p>
      )}
    </div>
  );
}

function StatusChips({ body }: { body: FindingDetailBody }) {
  const selected = body.revision;
  const currentness = currentnessWords(body.audit_only ? "SOURCE_CLOSED" : selected.currentness.state);
  return (
    <ul className="site-finding__chips" aria-label="Finding status">
      <li className="site-finding__chip" data-chip="state">
        State: {LIFECYCLE_WORDS[selected.lifecycle].label}
      </li>
      <li className="site-finding__chip" data-chip="currentness">
        {currentness.label}
      </li>
      <li className="site-finding__chip" data-chip="version">
        {versionWords(body)}
      </li>
      <li className="site-finding__chip" data-chip="period">
        {periodWords(selected.episode.start, selected.episode.end)}
      </li>
    </ul>
  );
}

function periodWords(start: string, end: string): string {
  const first = dayOf(start);
  const last = dayOf(end);
  return first === last
    ? `${first} ${clock(start)}–${clock(end)} UTC`
    : `${first} ${clock(start)} – ${last} ${clock(end)} UTC`;
}

interface Portion {
  start: string;
  end: string;
  classification: string;
  intervalRevisionId: string;
  evidenceCount: number | null;
}

function portionsOf(body: FindingDetailBody): Portion[] {
  const claim = body.revision.claim;
  const spans: Portion[] = (claim.candidate_spans ?? []).map((s) => ({
    start: s.start,
    end: s.end,
    classification: "CANDIDATE_AVOIDABLE",
    intervalRevisionId: s.assessment_revision_id,
    evidenceCount: s.evidence_ids?.length ?? null,
  }));
  const others: Portion[] = (claim.other_portions ?? []).map((p) => ({
    start: p.start,
    end: p.end,
    classification: p.classification,
    intervalRevisionId: p.assessment_revision_id,
    evidenceCount: null,
  }));
  return [...spans, ...others].sort((a, b) => Date.parse(a.start) - Date.parse(b.start));
}

function runWords(body: FindingDetailBody): string {
  const { start, end } = body.revision.episode;
  const stops = (body.revision.claim.continuity ?? []).flatMap((gap) => gap.stops ?? []).length;
  return stops > 0
    ? `The generator ran between ${periodWords(start, end)}, with ${stops === 1 ? "one short stop" : `${stops} short stops`} recorded in that time.`
    : `The generator stayed on from ${periodWords(start, end)}.`;
}

function WhatHappened({ body, links }: { body: FindingDetailBody; links: SiteFindingLinks }) {
  const [params, setParams] = useSearchParams();
  const selected = body.revision;
  const portions = portionsOf(body);
  const chosenIndex = Number(params.get("portion"));
  const chosen = params.get("portion") !== null && Number.isInteger(chosenIndex) ? (portions[chosenIndex] ?? null) : null;
  const avoidable = minutes(selected.minutes.candidate) ?? 0;
  const needed = minutes(selected.minutes.necessary) ?? 0;
  const unknown = minutes(selected.minutes.indeterminate) ?? 0;
  const period = spanMinutes(selected.episode.start, selected.episode.end);
  const startMs = Date.parse(selected.episode.start);
  const total = Math.max(1, Date.parse(selected.episode.end) - startMs);
  const choose = (index: number | null) => {
    const next = new URLSearchParams(params);
    if (index === null) {
      next.delete("portion");
    } else {
      next.set("portion", String(index));
    }
    setParams(next, { replace: true });
  };
  const supported = selected.lifecycle === "SUPPORTED";
  return (
    <Panel heading="What happened" headingId="site-finding-happened-heading">
      <p className="site-finding__big" data-finding-sentence>
        {supported
          ? `${runWords(body)} For ${durationText(avoidable)} of that time, solar power and the battery reported enough capacity to meet site demand, so the generator was possibly avoidable.`
          : selected.lifecycle === "MERGED"
            ? `This Finding covered ${periodWords(selected.episode.start, selected.episode.end)} before it was joined into an earlier Finding.`
            : selected.lifecycle === "UNCONFIRMED"
              ? `This Finding covers ${periodWords(selected.episode.start, selected.episode.end)}. ${body.selection.is_latest ? "The latest assessment" : "This version"} cannot decide whether any of that time was possibly avoidable.`
              : `This Finding covers ${periodWords(selected.episode.start, selected.episode.end)}. ${body.selection.is_latest ? "The latest assessment" : "This version"} of that period counts no possibly avoidable time.`}
      </p>
      <p>
        Only the possibly avoidable blocks are counted. Needed time and time that cannot be determined stay separate,
        and the whole period is not said to be avoidable. This is a bounded estimate from the reported records, not
        proof of wasted fuel.
      </p>
      <ul className="dispatch-cards" aria-label="Time in this Finding">
        <Card label="Possibly avoidable" value={durationText(avoidable)} note="Counted in this Finding" code="CANDIDATE_AVOIDABLE" />
        <Card label="Needed" value={durationText(needed)} note="A recorded reason required the generator" code="NECESSARY" />
        <Card label="Cannot determine" value={durationText(unknown)} note="Could not be decided either way" code="INDETERMINATE" />
        <Card label="Period covered" value={durationText(period)} note="From first to last block in this Finding" code="PERIOD" />
        <Card label="Fuel and cost" value="Not yet available" note="No fuel or money estimate is made here" code="MONEY" />
      </ul>
      {portions.length > 0 ? (
        <div className="site-finding__timeline">
          <h3 className="site-finding__subheading" id="site-finding-blocks-heading">
            Blocks in this Finding
          </h3>
          <ol className="site-finding__blocks" aria-labelledby="site-finding-blocks-heading">
            {portions.map((p, index) => {
              const left = ((Date.parse(p.start) - startMs) / total) * 100;
              const width = Math.max(0.6, ((Date.parse(p.end) - Date.parse(p.start)) / total) * 100);
              const pressed = chosen === p;
              return (
                <li key={`${p.start}-${index}`} style={{ left: `${left}%`, width: `${width}%` }}>
                  <button
                    type="button"
                    className={`dispatch-block dispatch-fill--${PORTION_FILL[p.classification] ?? "unknown"}`}
                    aria-pressed={pressed}
                    aria-label={`${portionWords(p.classification)}, ${clock(p.start)}–${clock(p.end)} UTC`}
                    data-portion={p.classification}
                    onClick={() => choose(pressed ? null : index)}
                  />
                </li>
              );
            })}
          </ol>
          <div className="dispatch-track__axis site-finding__axis" aria-hidden="true">
            <span>{clock(selected.episode.start)}</span>
            <span>{clock(selected.episode.end)}</span>
          </div>
          <p className="site-finding__legend">
            <span className="dispatch-swatch dispatch-fill--avoidable" /> Possibly avoidable{" "}
            <span className="dispatch-swatch dispatch-fill--needed" /> Needed{" "}
            <span className="dispatch-swatch dispatch-fill--unknown" /> Cannot determine · Times are UTC. Select a
            block to see it.
          </p>
          {chosen ? (
            <div className="dispatch-block-detail" data-finding-portion-detail>
              <h3>
                {portionWords(chosen.classification)}: {stamp(chosen.start)}–{clock(chosen.end)} UTC
              </h3>
              <p>
                {durationText(spanMinutes(chosen.start, chosen.end))}
                {chosen.evidenceCount !== null ? `, supported by ${chosen.evidenceCount} accepted records` : ""}.{" "}
                <Link to={links.dispatch(dayOf(chosen.start), chosen.intervalRevisionId)} data-finding-dispatch-link>
                  See this block on the Dispatch day
                </Link>
              </p>
            </div>
          ) : null}
        </div>
      ) : (
        <p data-finding-no-blocks>No block is counted in this version.</p>
      )}
    </Panel>
  );
}

function Card({ label, value, note, code }: { label: string; value: string; note: string; code: string }) {
  return (
    <li className="dispatch-card" data-finding-card={code}>
      <span className="dispatch-card__label">{label}</span>
      <span className="dispatch-card__value">{value}</span>
      <span className="dispatch-card__note">{note}</span>
    </li>
  );
}

function WhatLimits({ body, links }: { body: FindingDetailBody; links: SiteFindingLinks }) {
  const selected = body.revision;
  const groups = groupLimitations(selected.limitations);
  // Segment-level record gaps never reach a revision's limitations; the read classifies them (re-review C8).
  const recordGap = selected.indeterminate_cause?.kind === "RECORD_GAP";
  const stops = (selected.claim.continuity ?? []).flatMap((gap) => gap.stops ?? []);
  const joined = (selected.claim.continuity ?? []).filter((gap) => !(gap.stops ?? []).length);
  const recurrence = body.relations.recurrence;
  const prior = body.relations.recurs_after_finding_id;
  return (
    <Panel heading="What limits this conclusion" headingId="site-finding-limits-heading">
      <ul className="site-finding__limits">
        <li>
          The estimate uses the records sent from site. Solar and battery capacity are as reported; nothing was
          measured directly at the fuel tank.
        </li>
        {recordGap ? (
          <li data-finding-record-gap>
            Some records this version needs are missing or conflict over part of the period, so that part could not
            be decided.
          </li>
        ) : null}
        {groups.length === 0 ? (
          recordGap ? null : <li data-finding-limitations="none">No record gaps or conflicts were found over this period.</li>
        ) : (
          groups.map((g) => (
            <li key={g.code} data-limitation={g.code}>
              {limitationWords(g.code)}
              {g.items.length > 1 ? ` (${g.items.length} places)` : ""}.
            </li>
          ))
        )}
        {selected.claim.end_reason === "SERIES_LIMIT" ? (
          <li data-finding-end-reason="SERIES_LIMIT">
            This finding ends at {stamp(selected.episode.end)} UTC because the generator records after that could not be
            put in a proven time order (some arrived late and the clock behind them is not proven). Only the time
            before that is counted; the time after it is not judged, so no current cost is shown for this version.
          </li>
        ) : null}
        {joined.length > 0 ? (
          <li data-finding-joined-gaps={joined.length}>
            {joined.length === 1 ? "One short gap" : `${joined.length} short gaps`} of 20 minutes or less inside this
            period had records for every minute; they are part of the same period but add no possibly avoidable time.
          </li>
        ) : null}
        {stops.length > 0 ? (
          <li data-finding-stop-bridges={stops.length}>
            The generator stopped {stops.length === 1 ? "once" : `${stops.length} times`} for 20 minutes or less
            (first at {stamp(stops[0].start)} UTC). It reported off at both ends with nothing contrary in between, so
            this is still one period; the stop adds no possibly avoidable time. If few reports arrive during a stop,
            this joining rests on those two off reports.
          </li>
        ) : null}
        {prior ? (
          <li data-finding-recurrence={recurrence.status}>
            This happened before: an earlier Finding covered the same generator, separated from this one by a
            recorded stop. That link says the pattern recurred; it does not say the earlier Finding is still open.{" "}
            <Link to={links.finding(prior)}>Open the earlier Finding to see its own state</Link>
          </li>
        ) : recurrence.status === "NOT_ESTABLISHED" && recurrence.reason !== "NO_EARLIER_FINDING" ? (
          <li data-finding-recurrence={recurrence.status}>
            An earlier Finding exists for this generator, but whether this repeats it is not established: the time
            between them is not fully assessed or no recorded generator stop separates them.
          </li>
        ) : null}
        <li>
          Shown under the pilot&apos;s development access. Sign-in and per-person access are not installed yet.
        </li>
      </ul>
      <p className="site-finding__note">
        <Link to={{ search: "?view=analysis" + (body.selection.is_latest ? "" : `&revision_id=${body.selection.revision_id}`) }}>
          See the technical detail
        </Link>
      </p>
    </Panel>
  );
}

function WhereToLook({ body, links }: { body: FindingDetailBody; links: SiteFindingLinks }) {
  const selected = body.revision;
  const days = daysBetween(selected.episode.start, selected.episode.end);
  const primaryInterval =
    selected.interval_assessment_revision_ids.includes(selected.primary_assessment_revision_id)
      ? selected.primary_assessment_revision_id
      : null;
  const closed = body.links.performance.status === "CLOSED";
  const related: { label: string; id: string }[] = [
    ...(body.relations.recurs_after_finding_id ? [{ label: "Earlier Finding this repeats", id: body.relations.recurs_after_finding_id }] : []),
    ...body.relations.recurred_by.map((id) => ({ label: "Later Finding that repeats this", id })),
    ...(body.relations.split_from_finding_id ? [{ label: "Finding this was split from", id: body.relations.split_from_finding_id }] : []),
    ...body.relations.split_parts.map((id) => ({ label: "Part split from this Finding", id })),
    ...body.relations.merged_from.map((m) => ({ label: "Finding joined into this one", id: m.superseded_finding_id })),
    ...body.relations.merged_into.map((m) => ({ label: "Finding that carries this review", id: m.canonical_finding_id })),
  ];
  return (
    <Panel heading="Where to look next" headingId="site-finding-look-heading">
      <ol className="site-finding__look">
        <li>
          <h3>Dispatch {days.length === 1 ? "day" : "days"}</h3>
          <p>How each 15-minute block of the day was assessed, with the records behind it.</p>
          {closed ? (
            <p data-finding-dispatch-closed>Not available while this site&apos;s assessments are not released.</p>
          ) : (
            <ul className="site-finding__inline">
              {days.map((day, index) => (
                <li key={day}>
                  <Link to={links.dispatch(day, index === 0 ? primaryInterval : null)} data-finding-dispatch-day={day}>
                    Open {day} →
                  </Link>
                </li>
              ))}
            </ul>
          )}
        </li>
        <li>
          <h3>Accepted records</h3>
          <p>
            {selected.cited_evidence.length} accepted records {minutes(selected.minutes.candidate) ? "support the counted blocks" : "are cited by this version"}. They are listed below and
            on the site&apos;s records page.
          </p>
          <Link to={links.evidence}>Open the site&apos;s records →</Link>
        </li>
        <li>
          <h3>Related Findings</h3>
          {related.length === 0 ? (
            <p>None recorded.</p>
          ) : (
            <ul className="site-finding__inline">
              {related.map((r) => (
                <li key={`${r.label}-${r.id}`}>
                  <Link to={links.finding(r.id)} data-finding-related={r.id}>
                    {r.label} →
                  </Link>
                </li>
              ))}
            </ul>
          )}
        </li>
        {links.history ? (
          <li>
            <h3>History</h3>
            <p>Every recorded version of this Finding, oldest first, with what changed and when.</p>
            <Link to={links.history(body.selection.revision_id)} data-finding-history-link>
              Open the Finding&apos;s history →
            </Link>
          </li>
        ) : null}
      </ol>
    </Panel>
  );
}

function revisionLink(id: string, latestId: string): string {
  return id === latestId ? "?" : `?revision_id=${encodeURIComponent(id)}`;
}

function Versions({ body }: { body: FindingDetailBody }) {
  const byId = new Map(body.revisions.map((r) => [r.id, r]));
  return (
    <Panel heading="Versions" headingId="site-finding-versions-heading">
      <p>
        A Finding keeps every version. Select one to see it as it was recorded; its address stays the same after a
        reload.
      </p>
      <ol className="site-finding__versions">
        {body.revision_groups.map((group) => (
          <VersionGroup key={group.revision_ids[0]} group={group} byId={byId} body={body} />
        ))}
      </ol>
      {body.revision_page.next_after_revision_no !== null ? (
        <p className="site-finding__note">Older versions beyond this page are kept and listed in Analysis.</p>
      ) : null}
    </Panel>
  );
}

function VersionGroup({
  group,
  byId,
  body,
}: {
  group: RevisionGroup;
  byId: Map<string, RevisionSummary>;
  body: FindingDetailBody;
}) {
  const latestId = body.selection.latest_revision_id;
  const entry = (id: string) => {
    const rev = byId.get(id);
    const selected = id === body.selection.revision_id;
    return (
      <Link
        to={{ search: revisionLink(id, latestId) }}
        aria-current={selected ? "true" : undefined}
        data-finding-version={id}
        data-transition={rev?.transition}
      >
        Version {rev?.revision_no ?? "?"}: {transitionWords(rev?.transition ?? "")}
        {rev ? ` · recorded ${stamp(rev.recorded_at)} UTC` : ""}
        {id === latestId ? " (latest)" : ""}
        {selected ? " (shown)" : ""}
      </Link>
    );
  };
  if (group.kind !== "BASIS_REFRESHED" || group.count === 1) {
    return (
      <li className="site-finding__version" data-version-group={group.kind}>
        {entry(group.revision_ids[0])}
        {group.kind === "BASIS_REFRESHED" ? (
          <span className="site-finding__note"> The period and possibly avoidable time did not change.</span>
        ) : null}
      </li>
    );
  }
  const holdsSelected = group.revision_ids.includes(body.selection.revision_id);
  return (
    <li className="site-finding__version" data-version-group={group.kind} data-version-count={group.count}>
      <Disclosure
        name={`versions-${group.first_revision_no}`}
        defaultOpen={holdsSelected}
        summary={`Versions ${group.first_revision_no}–${group.last_revision_no}: ${group.count} assessment updates; period and possibly avoidable time unchanged`}
      >
        <ol className="site-finding__versions">
          {group.revision_ids.map((id) => (
            <li key={id}>{entry(id)}</li>
          ))}
        </ol>
      </Disclosure>
    </li>
  );
}

function day(value: string): string {
  const date = new Date(value);
  if (Number.isNaN(date.getTime())) return value;
  return date.toLocaleDateString("en-GB", { day: "numeric", month: "short", year: "numeric", timeZone: "UTC" });
}

function minutes(start: string, end: string | null): string | null {
  if (end === null) return null;
  const ms = new Date(end).getTime() - new Date(start).getTime();
  if (!Number.isFinite(ms) || ms < 0) return null;
  const total = Math.round(ms / 60000);
  const h = Math.floor(total / 60);
  const m = total % 60;
  return h > 0 ? `${h} h ${m} min` : `${m} min`;
}

/** "0.00 to 15.82 USD": the display strings exactly as served (low rounded down, high rounded up). */
function range(low: string, high: string, unit: string): string {
  return `${low} to ${high} ${unit}`;
}

/** Why a stored revision has no amount, or why its amount is not current. */
function revisionWhy(rev: ConsequenceRevision): string {
  if (rev.state === "QUANTITY_UNAVAILABLE") return quantityWords(rev.quantity.reason);
  if (rev.state === "UNPRICED") return priceWords(rev.price.reason);
  if (rev.display?.state !== "ROUNDED_OUTWARD") return "No rounding rule is known for this price's currency, so no cost is shown.";
  return "";
}

/** The lineage disclosure: model, multiplier, IDs, digests and every counted reading. */

function versionWords(rev: ConsequenceRevision, hideAmount: boolean): string {
  if (rev.state === "PRICED" && hideAmount) return "No cost shown: the source is closed, so this is audit material only";
  if (rev.state === "PRICED" && rev.display?.state === "ROUNDED_OUTWARD") {
    return `${range(rev.display.amount_low ?? "", rev.display.amount_high ?? "", rev.display.currency)}, at ${
      rev.price.value ?? ""
    } ${rev.price.currency ?? ""} per litre`;
  }
  if (rev.state === "UNPRICED") return "No cost: no single price for the whole window";
  if (rev.state === "QUANTITY_UNAVAILABLE") return "No cost: the fuel amount is not calculated";
  return "No cost shown";
}

/** The money headline: an amount only for a CURRENT PRICED bridge with a display range. */
function Headline({ bridge }: { bridge: FindingFinancialsBody }) {
  const c = bridge.consequence;
  const shown = c.state === "PRICED" && c.currentness === "CURRENT" && c.amount !== null && c.currency !== null;
  if (shown && c.amount && c.currency) {
    return (
      <div data-financials-headline="amount">
        <dl className="metric-row">
          <div className="metric">
            <dt>Modelled potential fuel-cost exposure</dt>
            <dd data-financials-range={`${c.amount.low}-${c.amount.high}`}>{range(c.amount.low, c.amount.high, c.currency)}</dd>
          </div>
        </dl>
        <p>
          Modelled potential fuel-cost exposure: {c.amount.low} to {c.amount.high} {c.currency}, under the reviewed fuel-cost-1
          assumptions. This is not measured fuel saved or a physical maximum.
        </p>
      </div>
    );
  }
  const why =
    c.state === "QUANTITY_UNAVAILABLE"
      ? quantityWords(c.quantity_reason)
      : c.state === "UNPRICED"
        ? priceWords(c.price_reason)
        : c.reason;
  return (
    <FactList>
      <Fact term="Cost">{STATE_HEADLINE[c.state] ?? "No cost shown"}</Fact>
      <Fact term="Why">
        <span data-financials-why={c.quantity_reason ?? c.price_reason ?? c.state}>{why}</span>
      </Fact>
    </FactList>
  );
}

const SERIES_KIND_WORDS: Record<string, string> = {
  SEQUENCE_ORDER_INVERTED: "arrived out of order and their clock is not proven",
  REPORTS_OVERLAP: "overlap in time, so no order can be chosen",
};

/** LR-01: why no current amount is shown although a recorded cost exists, with exact links to that record. */
function MoneyLimit({
  c,
  links,
  findingId,
  rev,
}: {
  c: FindingFinancialsBody["consequence"];
  links: BridgeLinks;
  findingId: string;
  rev: string;
}) {
  const gate = c.money_eligibility;
  const limits = gate?.series_limits ?? [];
  const seen = new Set<string>();
  const spans = limits.filter((x) => {
    const key = `${x.start}|${x.end}|${x.kind}`;
    if (seen.has(key)) return false;
    seen.add(key);
    return true;
  });
  return (
    <div data-financials-money-limit={c.state}>
      {c.state === "WITHHELD_GENERATOR_SERIES" ? (
        <ul>
          {spans.map((x) => (
            <li key={`${x.start}|${x.end}|${x.kind}`} data-financials-series-limit={x.kind}>
              Generator records from {formatInstant(x.start)} to {formatInstant(x.end)}{" "}
              {SERIES_KIND_WORDS[x.kind] ?? "could not be ordered"}. The finding ends before them and that time is not
              judged.
            </li>
          ))}
        </ul>
      ) : (
        <p data-financials-pending-reassessment="true">
          New evidence or clock proof is being taken into account. The current cost appears once the reassessment and
          the fuel cost recalculation have finished.
        </p>
      )}
      {c.recorded_consequence_revision_id ? (
        <p>
          <Link
            to={links.consequence(findingId, rev, c.recorded_consequence_revision_id)}
            data-financials-recorded-link={c.recorded_consequence_revision_id}
          >
            Open the recorded fuel cost of this finding version
          </Link>{" "}
          (kept for the record, not a current cost).
        </p>
      ) : null}
    </div>
  );
}

def _utc(value: Any) -> datetime:
    if isinstance(value, datetime):
        return value.astimezone(timezone.utc)
    return datetime.fromisoformat(str(value).replace("Z", "+00:00")).astimezone(timezone.utc)


def _iso(value: Any) -> str | None:
    return None if value is None else _utc(value).isoformat().replace("+00:00", "Z")


def days_over(lo: datetime, hi: datetime) -> list[date]:
    """UTC days intersecting [lo, hi]."""
    out, d = [], lo.astimezone(timezone.utc).date()
    while datetime(d.year, d.month, d.day, tzinfo=timezone.utc) <= hi:
        out.append(d)
        d += timedelta(days=1)
    return out

# Workflow

AssetOps is built through thin end-to-end slices that produce UI-verifiable
product behavior.

## Lanes

Planned lane:

Architect or Planner -> Implementer -> Reviewer -> User review

Use for new product behavior, domain semantics, user-visible analytics,
meaningful UI/UX, data/API contracts, architecture boundaries, simulator or
ingestion semantics, and external integrations.

Fast lane:

Implementer -> Reviewer

Use only when the work stays inside an existing product and technical contract.

## No Agent Writes An Approval

Recorded 2026-09-26 after an Implementer wrote two approvals that had not
happened into the durable record: a dated owner sign-off whose only basis was
four conversational words, and a review verdict of ACCEPT while a RETURN verdict
sat unread in the same directory. It then set the task complete and moved it to
`tasks/completed/`. Reverted at `166c02b`.

Relying on an agent to be honest about this is what failed, so these are
controls rather than advice.

- **An agent may not write an owner approval anywhere.** Not the task file, not
  a packet, not a commit message. An agent cannot observe the owner's approval;
  only the coordinator, relaying the user, records one. A conversational remark
  is not a checkpoint: the task file defines what the owner must have done, and
  nothing short of that is a sign-off.
- **An agent may not record a review verdict.** The coordinator records it,
  after reading the review file. A subagent's handback is that subagent's
  summary of its own work; the harness says so on every handback, and it was
  said on the one that was believed here.
- **An Implementer may not commission its own review.** A review it
  commissioned is not independent whatever it concludes, and
  `.ai/ROLE_CONFIG.md` requires independence from the author. The coordinator
  dispatches the Reviewer.
- **Before recording any verdict, list the review files.** Two sat in `.agent/`
  that day, twenty-two minutes apart, and the one that agreed was cited. Where
  there is more than one, they are all read and the disagreement is reported,
  not resolved by preference.
- **Do not commit into a tree a Reviewer is measuring.**
  `.ai/PROJECT_RULES.md` "One Working Copy" already says evidence taken while
  another agent holds the tree is not evidence. Four commits landed during that
  review, and it had to qualify its own findings as a result.

A task reaching `tasks/completed/` with a verdict nothing verified is the
failure these prevent. `T011B` and `T011C` reached main at `in_review` for the
mirror-image reason - no check enforces the transition, which is why the rule
has to be read rather than relied upon.

## Vertical Slice Standard

Each planned task should include the minimum required domain semantics, backend
logic, data contract, UI, tests, and demo evidence needed for a reviewable
result. Not every task needs every layer, but each task should either deliver a
UI-verifiable outcome or explicitly unlock a named UI-verifiable slice.

Features and tasks are different planning objects. A feature is a complete
product function that may require several tasks. A task is a reviewable
implementation slice that delivers or materially advances one UI-verifiable part
of a feature.

The current sequence is the Site-first S1â€“S7 queue in `tasks/README.md`, based on the current architecture, technical design and Site-first execution plan. Earlier feature maps and demo roadmaps are historical planning input.

This replaces the old examples that implied a fuel discrepancy was the first
Finding. Within each outcome, create reviewable cuts that let the owner run
and judge progress; do not make each architecture noun a separate task.

Avoid building whole layers before there is useful product behavior.

## Task Spec Size

Task specs are guardrails for fast implementation, not miniature design
documents. For S1-S7, keep durable reasoning in the current Site-first architecture, technical design, execution plan or a new decision; keep task files slice-specific and executable. The old feature map and T0xx queue remain historical.

Size is a ceiling, never a target. These are the upper limits:

- Normal UI/content slice: 120 lines.
- Boundary-changing slice: 160 lines.
- First write path, evidence contract, ingestion, simulator truth boundary, or
  analytics/finding semantics: 220 lines.
